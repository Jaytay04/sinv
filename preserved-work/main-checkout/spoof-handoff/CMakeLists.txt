cmake_minimum_required(VERSION 3.24)
project(client_machine_identifiers LANGUAGES C CXX)

if(NOT WIN32 OR NOT MSVC)
    message(FATAL_ERROR "This package requires Windows and the x64 MSVC toolchain.")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "This package requires an x64 build.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
# Existing fixture assertions must execute in the Release/static-CRT build.
string(REPLACE "/DNDEBUG" "" CMAKE_C_FLAGS_RELEASE "${CMAKE_C_FLAGS_RELEASE}")
string(REPLACE "/DNDEBUG" "" CMAKE_CXX_FLAGS_RELEASE "${CMAKE_CXX_FLAGS_RELEASE}")

set(DEVICE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/tools/hwid_spoofer")
set(INVENTORY_DIR "${CMAKE_CURRENT_SOURCE_DIR}/tools/efi_inventory")
function(client_options target)
    target_include_directories(${target} PRIVATE "${DEVICE_DIR}" "${DEVICE_DIR}/include" "${INVENTORY_DIR}/include")
    target_compile_definitions(${target} PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_compile_options(${target} PRIVATE /W4 /WX /utf-8)
endfunction()

add_library(client_identifier_engine STATIC "${DEVICE_DIR}/hwid_spoofer.c" "${DEVICE_DIR}/src/client_reads.cpp")
client_options(client_identifier_engine)
target_link_libraries(client_identifier_engine PUBLIC wbemuuid ole32 oleaut32)
add_executable(client_identifier_host_test "${DEVICE_DIR}/src/client_identifier_host_test.cpp")
client_options(client_identifier_host_test)
target_link_libraries(client_identifier_host_test PRIVATE client_identifier_engine)
