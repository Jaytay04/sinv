cmake_minimum_required(VERSION 3.24)
project(client_device_filters LANGUAGES C)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Windows x64 MSVC is required.")
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
separate_arguments(device_host_windows_libraries NATIVE_COMMAND "${CMAKE_C_STANDARD_LIBRARIES}")
set(CMAKE_C_STANDARD_LIBRARIES "")
string(REPLACE "/DNDEBUG" "" CMAKE_C_FLAGS_RELEASE "${CMAKE_C_FLAGS_RELEASE}")
option(DEVICE_FILTER_BUILD_KERNEL "Build the standalone WDM and NDIS filters" ON)
option(DEVICE_FILTER_ENABLE_REWRITE "Compile explicitly selected response rewrites" OFF)
set(DEVICE_FILTER_WDK_ROOT "C:/Program Files (x86)/Windows Kits/10" CACHE PATH "Installed WDK root")
set(DEVICE_FILTER_WDK_VERSION "10.0.28000.0" CACHE STRING "Installed WDK version")
set(DEVICE_FILTER_STORAGE_PROFILE_HEADER "" CACHE FILEPATH "Optional generated storage-only profile header")

if(DEVICE_FILTER_STORAGE_PROFILE_HEADER)
    get_filename_component(device_storage_profile_header "${DEVICE_FILTER_STORAGE_PROFILE_HEADER}" ABSOLUTE)
    if(NOT EXISTS "${device_storage_profile_header}")
        message(FATAL_ERROR "DEVICE_FILTER_STORAGE_PROFILE_HEADER does not exist.")
    endif()
    get_filename_component(device_storage_profile_directory "${device_storage_profile_header}" DIRECTORY)
    get_filename_component(device_storage_profile_filename "${device_storage_profile_header}" NAME)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(storage_serial_round_trip storage/test_storage_serial_rewrite.c storage/storage_serial_rewrite.c)
    add_executable(nic_address_round_trip nic/host_test.c nic/nic_policy.c)
    if(DEVICE_FILTER_STORAGE_PROFILE_HEADER)
        target_include_directories(storage_serial_round_trip PRIVATE "${device_storage_profile_directory}")
        target_compile_definitions(storage_serial_round_trip PRIVATE
            DEVICE_FILTER_STORAGE_PROFILE_HEADER=\"${device_storage_profile_filename}\")
    endif()
    foreach(target storage_serial_round_trip nic_address_round_trip)
        target_compile_options(${target} PRIVATE /W4 /WX /utf-8)
        target_link_libraries(${target} PRIVATE ${device_host_windows_libraries})
    endforeach()
    add_test(NAME storage_serial_round_trip COMMAND storage_serial_round_trip)
    add_test(NAME nic_address_round_trip COMMAND nic_address_round_trip)
endif()

if(DEVICE_FILTER_BUILD_KERNEL)
    set(wdk_include "${DEVICE_FILTER_WDK_ROOT}/Include/${DEVICE_FILTER_WDK_VERSION}")
    set(wdk_lib "${DEVICE_FILTER_WDK_ROOT}/Lib/${DEVICE_FILTER_WDK_VERSION}/km/x64")
    if(NOT EXISTS "${wdk_include}/km/ntddk.h" OR NOT EXISTS "${wdk_lib}/ndis.lib")
        message(FATAL_ERROR "The selected WDK is incomplete; kernel targets cannot be skipped.")
    endif()
    function(add_device_filter target)
        add_executable(${target} ${ARGN})
        set_target_properties(${target} PROPERTIES SUFFIX ".sys" MSVC_RUNTIME_LIBRARY "")
        target_include_directories(${target} PRIVATE "${wdk_include}/km" "${wdk_include}/shared")
        target_compile_definitions(${target} PRIVATE UNICODE _UNICODE _AMD64_ _WIN64 NTDDI_VERSION=0x0A00000C _WIN32_WINNT=0x0A00)
        # WDK structures intentionally pad explicitly aligned members.
        target_compile_options(${target} PRIVATE /kernel /GS /Gz /Zp8 /W4 /WX /wd4324 /utf-8 /Zl)
        target_link_directories(${target} PRIVATE "${wdk_lib}")
        target_link_options(${target} PRIVATE /DRIVER /SUBSYSTEM:NATIVE /ENTRY:GsDriverEntry /NODEFAULTLIB /MANIFEST:NO /INCREMENTAL:NO)
        target_link_libraries(${target} PRIVATE ntoskrnl.lib hal.lib BufferOverflowFastFailK.lib)
    endfunction()
    add_device_filter(client_storage_filter storage/storage_filter.c storage/storage_serial_rewrite.c)
    if(DEVICE_FILTER_STORAGE_PROFILE_HEADER)
        target_include_directories(client_storage_filter PRIVATE "${device_storage_profile_directory}")
        target_compile_definitions(client_storage_filter PRIVATE
            DEVICE_FILTER_STORAGE_PROFILE_HEADER=\"${device_storage_profile_filename}\")
    endif()
    add_device_filter(client_nic_filter nic/nic_filter.c nic/nic_policy.c)
    target_link_libraries(client_nic_filter PRIVATE ndis.lib)
    target_compile_definitions(client_nic_filter PRIVATE NDIS_SUPPORT_NDIS620=1)
    if(DEVICE_FILTER_ENABLE_REWRITE)
        target_compile_definitions(client_storage_filter PRIVATE DEVICE_FILTER_STORAGE_ENABLE_REWRITE=1)
        target_compile_definitions(client_nic_filter PRIVATE NIC_FILTER_SPOOF_ENABLED=1)
    else()
        target_compile_definitions(client_storage_filter PRIVATE DEVICE_FILTER_STORAGE_ENABLE_REWRITE=0)
        target_compile_definitions(client_nic_filter PRIVATE NIC_FILTER_SPOOF_ENABLED=0)
    endif()
endif()
